Componentes Principales
1. Webhook Handler (handlers/webhook.go)
- Endpoint: POST /webhook
- Recibe JSON de TradingView con formato:
{
  "symbol": "BTCUSDT",
  "action": "buy",  // o "sell"
  "price": 65000.50,
  "secret": "TU_SECRET_TOKEN"
}
- Valida el secret contra variable de entorno
- Compara con el último estado del usuario/símbolo
- Si cambió, envía alerta por Telegram
2. Bot Commands (handlers/commands.go)
- /start <chat_id>: Activa las alertas para ese usuario
- Registra el chat_id en storage
- Responde "Alertas activadas"
- /close: Desactiva las alertas
- Elimina del storage
- Responde "Alertas desactivadas"
3. Storage (storage/storage.go)
- Mapa en memoria: map[int64]*UserState
- UserState:
type UserState struct {
    ChatID      int64
    Active      bool
    LastSignals map[string]string  // symbol -> "buy"/"sell"
}
4. Telegram Service (services/telegram.go)
- Usa go-telegram-bot-api o llama directamente a la API HTTP
- Función SendMessage(chatID, text) para enviar alertas
Flujo de Datos
1. Usuario envía /start al bot → Se guarda su chat_id como activo
2. TradingView detecta cambio en Chandelier Exit → Envía webhook al servidor
3. Servidor recibe webhook, valida secret
4. Busca si hay usuarios activos suscritos a ese símbolo
5. Compara estado anterior vs nuevo ("buy"/"sell")
6. Si hay cambio, envía mensaje:

package models
import "sync"
type UserState struct {
    ChatID      int64
    Active      bool
    LastSignals map[string]string // symbol -> "buy" o "sell"
    mu          sync.RWMutex
}
type UserManager struct {
    users map[int64]*UserState
    mu    sync.RWMutex
}
func NewUserManager() *UserManager {
    return &UserManager{
        users: make(map[int64]*UserState),
    }
}
func (um *UserManager) Activate(chatID int64) {
    um.mu.Lock()
    defer um.mu.Unlock()
    if user, exists := um.users[chatID]; exists {
        user.Active = true
    } else {
        um.users[chatID] = &UserState{
            ChatID:      chatID,
            Active:      true,
            LastSignals: make(map[string]string),
        }
    }
}
func (um *UserManager) Deactivate(chatID int64) {
    um.mu.Lock()
    defer um.mu.Unlock()
    if user, exists := um.users[chatID]; exists {
        user.Active = false
    }
}
func (um *UserManager) GetActiveUsers() []*UserState {
    um.mu.RLock()
    defer um.mu.RWMUnlock()
    var active []*UserState
    for _, user := range um.users {
        if user.Active {
            active = append(active, user)
        }
    }
    return active
}
func (u *UserState) GetLastSignal(symbol string) string {
    u.mu.RLock()
    defer u.mu.RUnlock()
    return u.LastSignals[symbol]
}
func (u *UserState) SetLastSignal(symbol, signal string) {
    u.mu.Lock()
    defer u.mu.Unlock()
    u.LastSignals[symbol] = signal
}

package storage
import (
    "encoding/json"
    "os"
    "sync"
    "tradingview-bot/models"
)
type FileStorage struct {
    filePath string
    mu       sync.RWMutex
}
func NewFileStorage(filePath string) *FileStorage {
    return &FileStorage{filePath: filePath}
}
type PersistedData struct {
    Users map[int64]*UserStateData `json:"users"`
}
type UserStateData struct {
    ChatID      int64             `json:"chat_id"`
    Active      bool              `json:"active"`
    LastSignals map[string]string `json:"last_signals"`
}
func (fs *FileStorage) Load() map[int64]*models.UserState {
    fs.mu.RLock()
    defer fs.mu.RUnlock()
    data, err := os.ReadFile(fs.filePath)
    if err != nil {
        return make(map[int64]*models.UserState)
    }
    var persisted PersistedData
    if err := json.Unmarshal(data, &persisted); err != nil {
        return make(map[int64]*models.UserState)
    }
    users := make(map[int64]*models.UserState)
    for chatID, userData := range persisted.Users {
        users[chatID] = &models.UserState{
            ChatID:      userData.ChatID,
            Active:      userData.Active,
            LastSignals: userData.LastSignals,
        }
    }
    return users
}
func (fs *FileStorage) Save(users map[int64]*models.UserState) {
    fs.mu.Lock()
    defer fs.mu.Unlock()
    persisted := PersistedData{
        Users: make(map[int64]*UserStateData),
    }
    for chatID, user := range users {
        persisted.Users[chatID] = &UserStateData{
            ChatID:      user.ChatID,
            Active:      user.Active,
            LastSignals: user.LastSignals,
        }
    }
    data, err := json.MarshalIndent(persisted, "", "  ")
    if err != nil {
        return
    }
    os.WriteFile(fs.filePath, data, 0644)
}

package services
import (
    "fmt"
    "log"
    tgbotapi "github.com/go-telegram-bot-api/telegram-bot-api/v5"
)
type TelegramService struct {
    bot *tgbotapi.BotAPI
}
func NewTelegramService(token string) (*TelegramService, error) {
    bot, err := tgbotapi.NewBotAPI(token)
    if err != nil {
        return nil, err
    }
    log.Printf("Bot autorizado: %s", bot.Self.UserName)
    return &TelegramService{bot: bot}, nil
}
func (ts *TelegramService) SendMessage(chatID int64, text string) error {
    msg := tgbotapi.NewMessage(chatID, text)
    msg.ParseMode = "HTML"
    _, err := ts.bot.Send(msg)
    return err
}
func (ts *TelegramService) SendAlert(chatID int64, symbol, action, price string) {
    var emoji, signalText string
    switch action {
    case "buy":
        emoji = "🟢"
        signalText = "BUY"
    case "sell":
        emoji = "🔴"
        signalText = "SELL"
    default:
        emoji = "⚪"
        signalText = action
    }
    text := fmt.Sprintf(
        "%s <b>Alerta: %s</b>\n\nSeñal cambió a: <b>%s</b>\nPrecio: $%s",
        emoji, symbol, signalText, price,
    )
    if err := ts.SendMessage(chatID, text); err != nil {
        log.Printf("Error enviando alerta a %d: %v", chatID, err)
    }
}

package handlers
import (
    "fmt"
    "strings"
    "tradingview-bot/models"
    "tradingview-bot/services"
)
type CommandsHandler struct {
    userManager *models.UserManager
    telegram    *services.TelegramService
}
func NewCommandsHandler(um *models.UserManager, tg *services.TelegramService) *CommandsHandler {
    return &CommandsHandler{
        userManager: um,
        telegram:    tg,
    }
}
func (ch *CommandsHandler) HandleStart(chatID int64, args string) {
    parts := strings.Fields(args)
    if len(parts) < 1 {
        ch.telegram.SendMessage(chatID, 
            "uso: /start <tu_chat_id>\n\n "+
            "Para obtener tu chat_id, envía /myid al bot @userinfobot")
        return
    }
    var targetChatID int64
    fmt.Sscanf(parts[0], "%d", &targetChatID)
    if targetChatID == 0 {
        ch.telegram.SendMessage(chatID, "❌ Chat ID inválido")
        return
    }
    ch.userManager.Activate(targetChatID)
    ch.telegram.SendMessage(targetChatID, 
        "✅ <b>Alertas activadas</b>\n\n"+
        "Recibirás notificaciones cuando el indicador Chandelier Exit cambie de señal.\n\n"+
        "Para desactivar: /close")
    
    if targetChatID != chatID {
        ch.telegram.SendMessage(chatID, fmt.Sprintf("✅ Alertas activadas para chat ID: %d", targetChatID))
    }
}
func (ch *CommandsHandler) HandleClose(chatID int64) {
    ch.userManager.Deactivate(chatID)
    ch.telegram.SendMessage(chatID, 
        "❌ <b>Alertas desactivadas</b>\n\n"+
        "No recibirás más notificaciones.\n"+
        "Para reactivar: /start <tu_chat_id>")
}
func (ch *CommandsHandler) HandleMyID(chatID int64) {
    ch.telegram.SendMessage(chatID, fmt.Sprintf("Tu chat ID es: <code>%d</code>", chatID))
}
func (ch *CommandsHandler) HandleHelp(chatID int64) {
    text := "📚 <b>Comandos disponibles:</b>\n\n" +
        "/start <chat_id> - Activar alertas\n" +
        "/close - Desactivar alertas\n" +
        "/myid - Obtener tu chat ID\n" +
        "/help - Mostrar esta ayuda"
    ch.telegram.SendMessage(chatID, text)
}

package handlers
import (
    "encoding/json"
    "log"
    "net/http"
    "tradingview-bot/models"
    "tradingview-bot/services"
)
type WebhookPayload struct {
    Symbol string `json:"symbol"`
    Action string `json:"action"`
    Price  string `json:"price"`
    Secret string `json:"secret"`
}
type WebhookHandler struct {
    userManager *models.UserManager
    telegram    *services.TelegramService
    secret      string
}
func NewWebhookHandler(um *models.UserManager, tg *services.TelegramService, secret string) *WebhookHandler {
    return &WebhookHandler{
        userManager: um,
        telegram:    tg,
        secret:      secret,
    }
}
func (wh *WebhookHandler) HandleWebhook(w http.ResponseWriter, r *http.Request) {
    if r.Method != http.MethodPost {
        http.Error(w, "Método no permitido", http.StatusMethodNotAllowed)
        return
    }
    var payload WebhookPayload
    if err := json.NewDecoder(r.Body).Decode(&payload); err != nil {
        log.Printf("Error decodificando webhook: %v", err)
        http.Error(w, "JSON inválido", http.StatusBadRequest)
        return
    }
    if payload.Secret != wh.secret {
        log.Printf("Webhook rechazado: secret inválido")
        http.Error(w, "Unauthorized", http.StatusUnauthorized)
        return
    }
    if payload.Symbol == "" || payload.Action == "" {
        http.Error(w, "Campos requeridos: symbol, action", http.StatusBadRequest)
        return
    }
    log.Printf("Webhook recibido: %s %s @ %s", payload.Symbol, payload.Action, payload.Price)
    activeUsers := wh.userManager.GetActiveUsers()
    for _, user := range activeUsers {
        lastSignal := user.GetLastSignal(payload.Symbol)
        if lastSignal != payload.Action {
            user.SetLastSignal(payload.Symbol, payload.Action)
            wh.telegram.SendAlert(user.ChatID, payload.Symbol, payload.Action, payload.Price)
        }
    }
    w.Header().Set("Content-Type", "application/json")
    w.WriteHeader(http.StatusOK)
    json.NewEncoder(w).Encode(map[string]string{"status": "ok"})
}
8. main.go
package main
import (
    "log"
    "net/http"
    "os"
    "os/signal"
    "syscall"
    "tradingview-bot/config"
    "tradingview-bot/handlers"
    "tradingview-bot/models"
    "tradingview-bot/services"
    tgbotapi "github.com/go-telegram-bot-api/telegram-bot-api/v5"
)
func main() {
    cfg := config.Load()
    if cfg.TelegramBotToken == "" {
        log.Fatal("TELEGRAM_BOT_TOKEN no está configurado")
    }
    // Servicio de Telegram
    telegram, err := services.NewTelegramService(cfg.TelegramBotToken)
    if err != nil {
        log.Fatalf("Error inicializando Telegram: %v", err)
    }
    // Gestión de usuarios
    userManager := models.NewUserManager()
    // Handlers
    commandsHandler := handlers.NewCommandsHandler(userManager, telegram)
    webhookHandler := handlers.NewWebhookHandler(userManager, telegram, cfg.WebhookSecret)
    // Configurar bot de Telegram para polling
    bot, err := tgbotapi.NewBotAPI(cfg.TelegramBotToken)
    if err != nil {
        log.Fatalf("Error creando bot: %v", err)
    }
    u := tgbotapi.NewUpdate(0)
    u.Timeout = 60
    updates := bot.GetUpdatesChan(u)
    // Procesar comandos en goroutine
    go func() {
        for update := range updates {
            if update.Message == nil {
                continue
            }
            chatID := update.Message.Chat.ID
            text := update.Message.Text
            if !update.Message.IsCommand() {
                continue
            }
            switch update.Message.Command() {
            case "start":
                commandsHandler.HandleStart(chatID, update.Message.CommandArguments())
            case "close":
                commandsHandler.HandleClose(chatID)
            case "myid":
                commandsHandler.HandleMyID(chatID)
            case "help":
                commandsHandler.HandleHelp(chatID)
            default:
                telegram.SendMessage(chatID, "Comando no reconocido. Usa /help")
            }
        }
    }()
    // Servidor HTTP para webhooks
    http.HandleFunc("/webhook", webhookHandler.HandleWebhook)
    http.HandleFunc("/health", func(w http.ResponseWriter, r *http.Request) {
        w.WriteHeader(http.StatusOK)
        w.Write([]byte("OK"))
    })
    log.Printf("Bot iniciado: @%s", bot.Self.UserName)
    log.Printf("Servidor HTTP en puerto %s", cfg.Port)
    log.Println("Endpoints:")
    log.Println("  POST /webhook - Recibir alertas de TradingView")
    log.Println("  GET /health - Health check")
    // Manejar señal de apagado
    stop := make(chan os.Signal, 1)
    signal.Notify(stop, syscall.SIGINT, syscall.SIGTERM)
    go func() {
        if err := http.ListenAndServe(":"+cfg.Port, nil); err != nil {
            log.Fatalf("Error en servidor HTTP: %v", err)
        }
    }()
    <-stop
    log.Println("Apagando bot...")
    bot.StopReceivingUpdates()
}

package services
import (
    "bytes"
    "encoding/json"
    "fmt"
    "io"
    "net/http"

type LLMService struct {
    APIKey     string
    Provider   string // "openai", "anthropic", "deepseek"
    Model      string
    HTTPClient *http.Client
}
type ChatMessage struct {
    Role    string `json:"role"`
    Content string `json:"content"`
}
type OpenAIRequest struct {
    Model    string        `json:"model"`
    Messages []ChatMessage `json:"messages"`
    MaxTokens int          `json:"max_tokens"`
}
type OpenAIResponse struct {
    Choices []struct {
        Message struct {
            Content string `json:"content"`
        } `json:"message"`
    } `json:"choices"`
}
func NewLLMService(apiKey, provider, model string) *LLMService {
    return &LLMService{
        APIKey:     apiKey,
        Provider:   provider,
        Model:      model,
        HTTPClient: &http.Client{},
    }
}
const PINE_SCRIPT_SYSTEM_PROMPT = `Eres un experto en TradingView Pine Script v6. Tu tarea es:
1. ANALIZAR la estrategia de trading descrita en el texto
2. EXTRAER la lógica de trading (indicadores, condiciones de entrada/salida)
3. GENERAR código Pine Script v6 válido y funcional
REGLAS:
- Usa // @version=6
- Incluye strategy() con parámetros completos
- Añade alertcondition() para señales buy/sell
- Incluye inputs configurables
- Añade comentarios explicativos
- El código debe compilar sin errores en TradingView
RESPUESTA: Solo el código Pine Script, sin explicaciones adicionales.`
func (ls *LLMService) GeneratePineScript(strategyText string) (string, error) {
    userPrompt := fmt.Sprintf(`Analiza esta estrategia de trading y genera Pine Script v6:
---
%s
---
Genera el código Pine Script completo con alertas para buy/sell.`, strategyText)
    messages := []ChatMessage{
        {Role: "system", Content: PINE_SCRIPT_SYSTEM_PROMPT},
        {Role: "user", Content: userPrompt},
    }
    switch ls.Provider {
    case "openai":
        return ls.callOpenAI(messages)
    case "anthropic":
        return ls.callAnthropic(messages)
    case "deepseek":
        return ls.callDeepSeek(messages)
    default:
        return "", fmt.Errorf("proveedor no soportado: %s", ls.Provider)
    }
}
func (ls *LLMService) callOpenAI(messages []ChatMessage) (string, error) {
    reqBody := OpenAIRequest{
        Model:     ls.Model,
        Messages:  messages,
        MaxTokens: 2000,
    }
    
    jsonBody, _ := json.Marshal(reqBody)
    
    req, _ := http.NewRequest("POST", "https://api.openai.com/v1/chat/completions", 
        bytes.NewBuffer(jsonBody))
    req.Header.Set("Content-Type", "application/json")
    req.Header.Set("Authorization", "Bearer "+ls.APIKey)
    
    resp, err := ls.HTTPClient.Do(req)
    if err != nil {
        return "", err
    }
    defer resp.Body.Close()
    
    body, _ := io.ReadAll(resp.Body)
    
    var openAIResp OpenAIResponse
    if err := json.Unmarshal(body, &openAIResp); err != nil {
        return "", err
    }
    
    if len(openAIResp.Choices) == 0 {
        return "", fmt.Errorf("no se recibió respuesta del LLM")
    }
    
    return openAIResp.Choices[0].Message.Content, nil
}

package models
import (
    "sync"
    "time"
)
type Strategy struct {
    ID          string    `json:"id"`
    Name        string    `json:"name"`
    Description string    `json:"description"`
    PineScript  string    `json:"pine_script"`
    Active      bool      `json:"active"`
    CreatedAt   time.Time `json:"created_at"`
    Symbols     []string  `json:"symbols"` // Pares que monitorea
    mu          sync.RWMutex
}
type StrategyManager struct {
    strategies map[string]*Strategy
    mu         sync.RWMutex
}
func NewStrategyManager() *StrategyManager {
    return &StrategyManager{
        strategies: make(map[string]*Strategy),
    }
}
func (sm *StrategyManager) Create(name, description, pineScript string) *Strategy {
    sm.mu.Lock()
    defer sm.mu.Unlock()
    
    id := generateID()
    strategy := &Strategy{
        ID:          id,
        Name:        name,
        Description: description,
        PineScript:  pineScript,
        Active:      false,
        CreatedAt:   time.Now(),
        Symbols:     []string{},
    }
    
    sm.strategies[id] = strategy
    return strategy
}
func (sm *StrategyManager) Get(id string) *Strategy {
    sm.mu.RLock()
    defer sm.mu.RUnlock()
    return sm.strategies[id]
}
func (sm *StrategyManager) GetAll() []*Strategy {
    sm.mu.RLock()
    defer sm.mu.RUnlock()
    
    var strategies []*Strategy
    for _, s := range sm.strategies {
        strategies = append(strategies, s)
    }
    return strategies
}
func (sm *StrategyManager) Activate(id string) error {
    sm.mu.Lock()
    defer sm.mu.Unlock()
    
    if s, exists := sm.strategies[id]; exists {
        s.Active = true
        return nil
    }
    return fmt.Errorf("estrategia no encontrada: %s", id)
}
func (sm *StrategyManager) Deactivate(id string) error {
    sm.mu.Lock()
    defer sm.mu.Unlock()
    
    if s, exists := sm.strategies[id]; exists {
        s.Active = false
        return nil
    }
    return fmt.Errorf("estrategia no encontrada: %s", id)
}
func generateID() string {
    // Generar ID único
    b := make([]byte, 8)
    rand.Read(b)
    return hex.EncodeToString(b)
}

func (ch *CommandsHandler) HandleDocument(message *tgbotapi.Message) {
    chatID := message.Chat.ID
    document := message.Document
    
    if document == nil {
        ch.telegram.SendMessage(chatID, "❌ No se recibió ningún documento")
        return
    }
    
    // Validar que sea PDF
    if !strings.HasSuffix(document.FileName, ".pdf") {
        ch.telegram.SendMessage(chatID, "❌ Solo se aceptan archivos PDF")
        return
    }
    
    // Validar tamaño (20MB max)
    if document.FileSize > 20*1024*1024 {
        ch.telegram.SendMessage(chatID, "❌ El archivo excede el límite de 20MB")
        return
    }
    
    // Descargar archivo
    ch.telegram.SendMessage(chatID, "📥 Descargando PDF...")
    
    fileConfig := tgbotapi.FileConfig{FileID: document.FileID}
    file, err := ch.bot.GetFile(fileConfig)
    if err != nil {
        ch.telegram.SendMessage(chatID, "❌ Error descargando el archivo")
        return
    }
    
    // Crear directorio de uploads
    os.MkdirAll(ch.uploadDir, 0755)
    
    // Guardar archivo
    filePath := filepath.Join(ch.uploadDir, fmt.Sprintf("%d_%s", chatID, document.FileName))
    out, err := os.Create(filePath)
    if err != nil {
        ch.telegram.SendMessage(chatID, "❌ Error guardando el archivo")
        return
    }
    defer out.Close()
    
    // URL de descarga de Telegram
    fileURL := fmt.Sprintf("https://api.telegram.org/file/bot%s/%s", 
        ch.bot.Token, file.FilePath)
    
    resp, err := http.Get(fileURL)
    if err != nil {
        ch.telegram.SendMessage(chatID, "❌ Error descargando contenido")
        return
    }
    defer resp.Body.Close()
    
    io.Copy(out, resp.Body)
    
    // Extraer texto del PDF
    ch.telegram.SendMessage(chatID, "🔍 Analizando PDF...")
    
    text, err := ch.pdfExtractor.ExtractText(filePath)
    if err != nil {
        ch.telegram.SendMessage(chatID, fmt.Sprintf("❌ Error extrayendo texto: %v", err))
        return
    }
    
    if len(text) < 100 {
        ch.telegram.SendMessage(chatID, "❌ El PDF no contiene suficiente texto legible")
        return
    }
    
    // Generar Pine Script con LLM
    ch.telegram.SendMessage(chatID, "🧠 Generando estrategia con IA...")
    
    pineScript, err := ch.llmService.GeneratePineScript(text)
    if err != nil {
        ch.telegram.SendMessage(chatID, fmt.Sprintf("❌ Error generando código: %v", err))
        return
    }
    
    // Guardar estrategia
    strategy := ch.strategyManager.Create(
        document.FileName,
        fmt.Sprintf("Estrategia extraída de %s", document.FileName),
        pineScript,
    )
    
    // Enviar código al usuario
    response := fmt.Sprintf(
        "✅ <b>Estrategia generada exitosamente</b>\n\n"+
        "📄 <b>Archivo:</b> %s\n"+
        "🆔 <b>ID:</b> <code>%s</code>\n\n"+
        "📝 <b>Código Pine Script:</b>\n\n"+
        "<code>%s</code>\n\n"+
        "Para usar esta estrategia:\n"+
        "1. Copia el código\n"+
        "2. Pégalo en TradingView (Pine Editor)\n"+
        "3. Añade alertas con webhook a tu bot\n"+
        "4. Activa la estrategia: /strategy %s",
        document.FileName, strategy.ID, pineScript, strategy.ID,
    )
    
    ch.telegram.SendMessage(chatID, response)
    
    // Limpiar archivo
    os.Remove(filePath)
}
func (ch *CommandsHandler) HandleStrategies(chatID int64) {
    strategies := ch.strategyManager.GetAll()
    
    if len(strategies) == 0 {
        ch.telegram.SendMessage(chatID, 
            "📭 No hay estrategias aprendidas aún.\n\n"+
            "Usa /learn y envía un PDF para empezar.")
        return
    }
    
    text := "📋 <b>Estrategias aprendidas:</b>\n\n"
    for _, s := range strategies {
        status := "❌ Inactiva"
        if s.Active {
            status = "✅ Activa"
        }
        text += fmt.Sprintf(
            "🆔 <code>%s</code>\n"+
            "📄 %s\n"+
            "Estado: %s\n\n",
            s.ID, s.Name, status,
        )
    }
    
    text += "Para activar/desactivar: /strategy <id>"
    ch.telegram.SendMessage(chatID, text)
}
func (ch *CommandsHandler) HandleStrategy(chatID int64, args string) {
    parts := strings.Fields(args)
    if len(parts) < 1 {
        ch.telegram.SendMessage(chatID, "uso: /strategy <id>")
        return
    }
    
    strategyID := parts[0]
    strategy := ch.strategyManager.Get(strategyID)
    
    if strategy == nil {
        ch.telegram.SendMessage(chatID, fmt.Sprintf("❌ Estrategia no encontrada: %s", strategyID))
        return
    }
    
    // Toggle active state
    if strategy.Active {
        ch.strategyManager.Deactivate(strategyID)
        ch.telegram.SendMessage(chatID, fmt.Sprintf("❌ Estrategia <code>%s</code> desactivada", strategyID))
    } else {
        ch.strategyManager.Activate(strategyID)
        ch.telegram.SendMessage(chatID, 
            fmt.Sprintf("✅ Estrategia <code>%s</code> activada\n\n"+
                "Ahora recibirás alertas cuando la estrategia genere señales.\n"+
                "Asegúrate de tener alertas configuradas en TradingView con webhook a tu bot.", 
                strategyID))
    }
}

package main
import (
    "log"
    "net/http"
    "os"
    "os/signal"
    "syscall"
    "tradingview-bot/config"
    "tradingview-bot/handlers"
    "tradingview-bot/models"
    "tradingview-bot/services"
    tgbotapi "github.com/go-telegram-bot-api/telegram-bot-api/v5"
)
func main() {
    cfg := config.Load()
    if cfg.TelegramBotToken == "" {
        log.Fatal("TELEGRAM_BOT_TOKEN no está configurado")
    }
    if cfg.LLMAPIKey == "" {
        log.Fatal("LLM_API_KEY no está configurado (OpenAI/Anthropic/DeepSeek)")
    }
    // Servicios
    telegram, err := services.NewTelegramService(cfg.TelegramBotToken)
    if err != nil {
        log.Fatalf("Error inicializando Telegram: %v", err)
    }
    pdfExtractor := services.NewPDFExtractor()
    llmService := services.NewLLMService(cfg.LLMAPIKey, cfg.LLMProvider, cfg.LLMModel)
    // Gestión
    userManager := models.NewUserManager()
    strategyManager := models.NewStrategyManager()
    // Bot de Telegram
    bot, err := tgbotapi.NewBotAPI(cfg.TelegramBotToken)
    if err != nil {
        log.Fatalf("Error creando bot: %v", err)
    }
    // Handlers
    commandsHandler := handlers.NewCommandsHandler(
        userManager, strategyManager, telegram, 
        pdfExtractor, llmService, bot,
    )
    webhookHandler := handlers.NewWebhookHandler(userManager, strategyManager, telegram, cfg.WebhookSecret)
    // Polling de Telegram
    u := tgbotapi.NewUpdate(0)
    u.Timeout = 60
    updates := bot.GetUpdatesChan(u)
    go func() {
        for update := range updates {
            if update.Message == nil {
                continue
            }
            chatID := update.Message.Chat.ID
            // Manejar documentos (PDFs)
            if update.Message.Document != nil {
                commandsHandler.HandleDocument(update.Message)
                continue
            }
            if !update.Message.IsCommand() {
                continue
            }
            switch update.Message.Command() {
            case "start":
                commandsHandler.HandleStart(chatID, update.Message.CommandArguments())
            case "close":
                commandsHandler.HandleClose(chatID)
            case "myid":
                commandsHandler.HandleMyID(chatID)
            case "help":
                commandsHandler.HandleHelp(chatID)
            case "learn":
                commandsHandler.HandleLearn(chatID)
            case "strategies":
                commandsHandler.HandleStrategies(chatID)
            case "strategy":
                commandsHandler.HandleStrategy(chatID, update.Message.CommandArguments())
            default:
                telegram.SendMessage(chatID, "Comando no reconocido. Usa /help")
            }
        }
    }()
    // Servidor HTTP para webhooks
    http.HandleFunc("/webhook", webhookHandler.HandleWebhook)
    http.HandleFunc("/health", func(w http.ResponseWriter, r *http.Request) {
        w.WriteHeader(http.StatusOK)
        w.Write([]byte("OK"))
    })
    log.Printf("Bot iniciado: @%s", bot.Self.UserName)
    log.Printf("Servidor HTTP en puerto %s", cfg.Port)
    log.Println("Endpoints:")
    log.Println("  POST /webhook - Recibir alertas de TradingView")
    log.Println("  GET /health - Health check")
    stop := make(chan os.Signal, 1)
    signal.Notify(stop, syscall.SIGINT, syscall.SIGTERM)
    go func() {
        if err := http.ListenAndServe(":"+cfg.Port, nil); err != nil {
            log.Fatalf("Error en servidor HTTP: %v", err)
        }
    }()
    <-stop
    log.Println("Apagando bot...")
    bot.StopReceivingUpdates()
}

func getEnv(key, fallback string) string {
    if value, exists := os.LookupEnv(key); exists {
        return value
    }
    return fallback
}
